cmake_minimum_required(VERSION 3.24)
project(TeardownVRCore VERSION 0.1.0 LANGUAGES CXX)
enable_testing()
add_library(tdvr_math SHARED native/src/math.cpp)
target_include_directories(tdvr_math PUBLIC native/include)
target_compile_features(tdvr_math PUBLIC cxx_std_17)
target_compile_definitions(tdvr_math PRIVATE TDVR_BUILD)
if(MSVC)
  target_compile_options(tdvr_math PRIVATE /W4 /WX /permissive-)
endif()
add_executable(tdvr_math_tests native/tests/math_tests.cpp)
target_link_libraries(tdvr_math_tests PRIVATE tdvr_math)
add_test(NAME pose_and_projection COMMAND tdvr_math_tests)
